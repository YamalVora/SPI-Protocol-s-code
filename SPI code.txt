module spi_sim (
    input clk,
    input rst,
    input sclk,
    input cs,
    input hdi,
    input cpol,
    input cpha,
    input [7:0] data_in,

    output hdo,
    output reg [3:0] clk_count,
    output reg data_valid,
    output reg [7:0] shift_reg,
    output reg [7:0] data_out,
    output reg [7:0] tx_shift_reg
);

   
    reg [2:0] sclk_sync;
    reg [1:0] cs_sync;
    reg [1:0] mosi_sync;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            sclk_sync <= 3'b000;
            cs_sync   <= 2'b11;
            mosi_sync <= 2'b00;
        end
        else begin
            sclk_sync <= {sclk_sync[1:0], sclk};
            cs_sync   <= {cs_sync[0], cs};
            mosi_sync <= {mosi_sync[0], hdi};
        end
    end

    wire rise = (sclk_sync[2:1] == 2'b01);
    wire fall = (sclk_sync[2:1] == 2'b10);

    
    wire leading_edge  = cpol ? fall : rise;
    wire trailing_edge = cpol ? rise : fall;


    wire sample_edge = cpha ? trailing_edge : leading_edge;
    wire change_edge = cpha ? leading_edge  : trailing_edge;

   
    assign hdo = (!cs_sync[1]) ? tx_shift_reg[7] : 1'b0;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            clk_count    <= 4'd0;
            shift_reg    <= 8'd0;
            data_out     <= 8'd0;
            data_valid   <= 1'b0;
            tx_shift_reg <= 8'd0;
        end
        else if (cs_sync[1]) begin
            
            clk_count    <= 4'd0;
            shift_reg    <= 8'd0;
            data_valid   <= 1'b0;
            tx_shift_reg <= data_in;
        end
        else begin
            data_valid <= 1'b0;

            
            if (sample_edge) begin
                shift_reg <= {shift_reg[6:0], mosi_sync[1]};

                if (clk_count == 4'd7) begin
                    data_out   <= {shift_reg[6:0], mosi_sync[1]};
                    data_valid <= 1'b1;
                    clk_count  <= 4'd0;
                end
                else begin
                    clk_count <= clk_count + 1'b1;
                end
            end

           
            if (change_edge && clk_count != 4'd0) begin
                tx_shift_reg <= {tx_shift_reg[6:0], 1'b0};
            end
        end
    end

endmodule
