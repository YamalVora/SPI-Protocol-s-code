`timescale 1ns / 1ps

module spi_simtb;

    reg clk;
    reg rst;
    reg sclk;
    reg cs;
    reg hdi;
    reg cpol;
    reg cpha;
    reg [7:0] data_in;

    wire hdo;
    wire [3:0] clk_count;
    wire data_valid;
    wire [7:0] shift_reg;
    wire [7:0] data_out;
    wire [7:0] tx_shift_reg;

    reg [7:0] captured_data;
    reg [7:0] master_rx;

    integer valid_count;
    integer rx_errors;
    integer tx_errors;

    spi_sim uut (
        .clk(clk),
        .rst(rst),
        .sclk(sclk),
        .cs(cs),
        .hdi(hdi),
        .cpol(cpol),
        .cpha(cpha),
        .data_in(data_in),
        .hdo(hdo),
        .clk_count(clk_count),
        .data_valid(data_valid),
        .shift_reg(shift_reg),
        .data_out(data_out),
        .tx_shift_reg(tx_shift_reg)
    );

   
    initial begin
        clk = 0;
        forever #5 clk = ~clk;
    end

   
    always @(posedge clk) begin
        if (data_valid) begin
            captured_data <= data_out;
            valid_count <= valid_count + 1;
        end
    end

  
    task send_byte;
        input [1:0] mode;
        input [7:0] master_tx;

        integer i;
        integer previous_count;

        begin
            
            cs = 1;
            cpol = mode[1];
            cpha = mode[0];

            sclk = cpol;
            hdi = 0;
            data_in = 8'hA5;

            
            #200;

            previous_count = valid_count;
            master_rx = 8'h00;

            cs = 0;
            #100;
            for (i = 7; i >= 0; i = i - 1) begin

                if (cpha == 0) begin
                    
                    hdi = master_tx[i];
                    #100;

                    // Leading edge: sample MOSI and MISO
                    sclk = ~cpol;
                    #50;

                    master_rx = {master_rx[6:0], hdo};

                    #50;

                    // Trailing edge: advance TX shift register
                    sclk = cpol;
                    #100;

                end
                else begin

                    // Leading edge: prepare TX bit
                    sclk = ~cpol;
                    #50;

                    // Put MOSI bit before trailing edge
                    hdi = master_tx[i];
                    #100;

                    // Trailing edge: sample MOSI and MISO
                    sclk = cpol;
                    #50;

                    master_rx = {master_rx[6:0], hdo};

                    #50;
                end
            end

        
            #100;

            // Check FPGA reception (MOSI)
            if ((valid_count != previous_count + 1) ||
                (captured_data !== master_tx)) begin

                rx_errors = rx_errors + 1;

                $display(
                    "Mode %0d RX FAIL: Expected=%h Got=%h",
                    mode, master_tx, captured_data
                );
            end
            else begin
                $display(
                    "Mode %0d RX PASS: %h",
                    mode, captured_data
                );
            end

            // Check FPGA transmission (MISO)
            if (master_rx !== 8'hA5) begin

                tx_errors = tx_errors + 1;

                $display(
                    "Mode %0d TX FAIL: Expected=A5 Got=%h",
                    mode, master_rx
                );
            end
            else begin
                $display(
                    "Mode %0d TX PASS: %h",
                    mode, master_rx
                );
            end

            cs = 1;
            #200;
        end
    endtask

    initial begin
        $dumpfile("spi_simtb.vcd");
        $dumpvars(0, spi_simtb);

        rst = 1;
        cs = 1;
        sclk = 0;
        hdi = 0;
        cpol = 0;
        cpha = 0;
        data_in = 8'hA5;

        captured_data = 0;
        master_rx = 0;
        valid_count = 0;
        rx_errors = 0;
        tx_errors = 0;

        #100;
        rst = 0;

        // Test all four SPI modes
        send_byte(2'd0, 8'hAA);
        send_byte(2'd1, 8'hF0);
        send_byte(2'd2, 8'h0F);
        send_byte(2'd3, 8'hCC);

        $finish;
    end

endmodule
