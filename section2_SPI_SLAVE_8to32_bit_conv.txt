module spi_slave_8to32 #(
    parameter CPOL = 0,
    parameter CPHA = 0
)(
    input             clk,          
    input             rst,

  
    input             sclk,
    input             cs,        
    input             mosi,
    output            miso,


    output reg [31:0] word_out,
    output reg        word_valid,   
    input             word_ready,  

    // Debug / status
    output reg [7:0]  byte_out,     
    output reg        byte_valid,   
    output reg        overflow      
);

    assign miso = 1'b0;   

  
    reg [2:0] sclk_sync;
    reg [1:0] cs_sync;
    reg [1:0] mosi_sync;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            sclk_sync <= {3{(CPOL != 0)}};   
            cs_sync   <= 2'b11;
            mosi_sync <= 2'b00;
        end
        else begin
            sclk_sync <= {sclk_sync[1:0], sclk};
            cs_sync   <= {cs_sync[0], cs};
            mosi_sync <= {mosi_sync[0], mosi};
        end
    end

  
    wire rise = (sclk_sync[2:1] == 2'b01);
    wire fall = (sclk_sync[2:1] == 2'b10);

    wire leading_edge  = (CPOL != 0) ? fall : rise;
    wire trailing_edge = (CPOL != 0) ? rise : fall;
    wire sample_edge   = (CPHA != 0) ? trailing_edge : leading_edge;


    reg  [2:0]  bit_cnt;     
    reg  [1:0]  byte_cnt;    
    reg  [7:0]  bit_shift;
    reg  [23:0] word_shift;  

    wire [7:0]  new_byte = {bit_shift[6:0], mosi_sync[1]};

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            bit_cnt    <= 3'd0;
            byte_cnt   <= 2'd0;
            bit_shift  <= 8'd0;
            word_shift <= 24'd0;
            word_out   <= 32'd0;
            word_valid <= 1'b0;
            byte_out   <= 8'd0;
            byte_valid <= 1'b0;
            overflow   <= 1'b0;
        end
        else begin
            byte_valid <= 1'b0;
            if (word_valid && word_ready)
                word_valid <= 1'b0;

            if (cs_sync[1]) begin
                bit_cnt   <= 3'd0;
                byte_cnt  <= 2'd0;
                bit_shift <= 8'd0;
            end
            else if (sample_edge) begin
                bit_shift <= new_byte;

                if (bit_cnt == 3'd7) begin
                    bit_cnt    <= 3'd0;
                    byte_out   <= new_byte;
                    byte_valid <= 1'b1;

                    if (byte_cnt == 2'd3) begin
                        byte_cnt <= 2'd0;
                        if (word_valid && !word_ready)
                            overflow <= 1'b1;           
                        else begin
                            word_out   <= {word_shift, new_byte};
                            word_valid <= 1'b1;
                        end
                    end
                    else begin
                        word_shift <= {word_shift[15:0], new_byte};
                        byte_cnt   <= byte_cnt + 1'b1;
                    end
                end
                else
                    bit_cnt <= bit_cnt + 1'b1;
            end
        end
    end
endmodule
